import { http } from '@/shared/api'
import type { User, CreateUserDto, UpdateUserDto } from '@/entities/user'
import type {
    Readable,
    Creatable,
    Editable,
    Deletable,
    PaginatedResult,
} from '@/shared/api'

interface UsersResponse {
    users: User[]
    total: number
    skip: number
    limit: number
}

const STORAGE_MODIFIED_USERS = 'local_modified_users'
const STORAGE_CREATED_USERS = 'local_created_users'
const STORAGE_DELETED_USERS = 'local_deleted_users'

function getSavedObject<T>(key: string, fallback: T): T {
    try {
        const data = localStorage.getItem(key)
        return data ? JSON.parse(data) : fallback
    } catch {
        return fallback
    }
}

export class UserService
    implements
        Readable<User>,
        Creatable<User, CreateUserDto>,
        Editable<User, UpdateUserDto>,
        Deletable<User>
{
    private modifiedUsers: Map<number, User>
    private createdUsers: User[]
    private deletedUserIds: Set<number>

    constructor() {
        this.modifiedUsers = new Map()
        const savedMods = getSavedObject<Record<number, User>>(STORAGE_MODIFIED_USERS, {})
        Object.entries(savedMods).forEach(([id, user]) => {
            this.modifiedUsers.set(Number(id), user)
        })

        this.createdUsers = getSavedObject<User[]>(STORAGE_CREATED_USERS, [])
        this.deletedUserIds = new Set(getSavedObject<number[]>(STORAGE_DELETED_USERS, []))
    }

    private saveState() {
        const modsObj: Record<number, User> = {}
        this.modifiedUsers.forEach((user, id) => {
            const copy = { ...user } as Record<string, unknown>
            delete copy['password']
            modsObj[id] = copy as unknown as User
        })

        const cleanedCreatedUsers = this.createdUsers.map(user => {
            const copy = { ...user } as Record<string, unknown>
            delete copy['password']
            return copy as unknown as User
        })

        localStorage.setItem(STORAGE_MODIFIED_USERS, JSON.stringify(modsObj))
        localStorage.setItem(STORAGE_CREATED_USERS, JSON.stringify(cleanedCreatedUsers))
        localStorage.setItem(STORAGE_DELETED_USERS, JSON.stringify(Array.from(this.deletedUserIds)))
    }

    async getAll(params: { limit?: number; skip?: number; q?: string } = {}): Promise<PaginatedResult<User>> {
        const query = new URLSearchParams()

        if (params.limit !== undefined) query.set('limit', String(params.limit))
        if (params.skip !== undefined) query.set('skip', String(params.skip))
        if (params.q) query.set('q', params.q)

        const endpoint = params.q ? '/users/search' : '/users'
        const suffix = query.size > 0 ? `?${query.toString()}` : ''
        const response = await http<UsersResponse>(`${endpoint}${suffix}`)

        let usersList = response.users
            .filter(u => !this.deletedUserIds.has(u.id))
            .map(u => this.modifiedUsers.get(u.id) || u)

        const matchingCreated = this.createdUsers.filter(u => {
            if (this.deletedUserIds.has(u.id)) return false
            if (!params.q) return true
            const qLower = params.q.toLowerCase()
            return (
                u.firstName.toLowerCase().includes(qLower) ||
                u.lastName.toLowerCase().includes(qLower) ||
                u.email.toLowerCase().includes(qLower) ||
                u.username.toLowerCase().includes(qLower)
            )
        })

        if (params.skip === 0 || !params.skip) {
            usersList = [...matchingCreated, ...usersList]
        }

        const deletedCount = params.q
            ? response.users.filter(u => this.deletedUserIds.has(u.id)).length
            : this.deletedUserIds.size

        const calculatedTotal = response.total + matchingCreated.length - deletedCount

        return {
            users: usersList,
            total: Math.max(0, calculatedTotal)
        }
    }

    async getById(id: number): Promise<User> {
        if (this.deletedUserIds.has(id)) {
            throw new Error('Користувача було видалено')
        }

        const created = this.createdUsers.find(u => u.id === id)
        if (created) return this.modifiedUsers.get(id) || created

        if (this.modifiedUsers.has(id)) {
            return this.modifiedUsers.get(id)!
        }

        const user = await http<User>(`/users/${id}`)
        return this.modifiedUsers.get(id) || user
    }

    async create(data: CreateUserDto): Promise<User> {
        const dataWithoutPassword: Record<string, unknown> = { ...data }
        delete dataWithoutPassword['password']

        const apiResult = await http<User>('/users/add', {
            method: 'POST',
            body: JSON.stringify(data),
        })

        const newId = Date.now()
        const newUser: User = {
            ...dataWithoutPassword,
            ...apiResult,
            id: newId
        } as unknown as User

        this.createdUsers.unshift(newUser)
        this.saveState()

        return newUser
    }

    async update(id: number, data: UpdateUserDto): Promise<User> {
        const dataWithoutPassword: Record<string, unknown> = { ...data }
        delete dataWithoutPassword['password']

        const apiResult = await http<User>(`/users/${id}`, {
            method: 'PUT',
            body: JSON.stringify(data),
        })

        const existing = await this.getById(id)
        const updatedUser: User = {
            ...existing,
            ...apiResult,
            ...dataWithoutPassword,
            id
        } as unknown as User

        this.modifiedUsers.set(id, updatedUser)

        const createdIdx = this.createdUsers.findIndex(u => u.id === id)
        if (createdIdx !== -1) {
            this.createdUsers[createdIdx] = updatedUser
        }

        this.saveState()
        return updatedUser
    }

    async delete(id: number): Promise<void> {
        await http<unknown>(`/users/${id}`, {
            method: 'DELETE',
        })

        const wasCreated = this.createdUsers.some(u => u.id === id)
        this.createdUsers = this.createdUsers.filter(u => u.id !== id)
        if (!wasCreated) {
            this.deletedUserIds.add(id)
        }

        this.modifiedUsers.delete(id)
        this.saveState()
    }
}

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { AuthUser, LoginCredentials } from '@/entities/session'
import type { User } from '@/entities/user'
import { AuthService } from '@/entities/session'

export const useAuthStore = defineStore('auth', () => {
    const user = ref<AuthUser | null>(
        JSON.parse(localStorage.getItem('auth_user') || 'null')
    )
    const accessToken = ref<string | null>(
        localStorage.getItem('access_token')
    )
    const refreshToken = ref<string | null>(
        localStorage.getItem('refresh_token')
    )

    const isAuthenticated = computed(() => Boolean(accessToken.value))

    function setTokens(access: string, refresh: string) {
        accessToken.value = access
        refreshToken.value = refresh
        localStorage.setItem('access_token', access)
        localStorage.setItem('refresh_token', refresh)
    }

    function setUser(userData: AuthUser | User) {
        user.value = userData as AuthUser
        localStorage.setItem('auth_user', JSON.stringify(userData))
    }

    function clearAuth() {
        user.value = null
        accessToken.value = null
        refreshToken.value = null
        localStorage.removeItem('access_token')
        localStorage.removeItem('refresh_token')
        localStorage.removeItem('auth_user')
    }

    async function login(credentials: LoginCredentials) {
        const authService = new AuthService()
        const response = await authService.login(credentials)
        const { accessToken: access, refreshToken: refresh, ...userData } = response

        setTokens(access, refresh)
        setUser(userData)
        return response
    }

    async function fetchCurrentUser() {
        if (!accessToken.value) return
        try {
            const authService = new AuthService()
            const currentUser = await authService.getCurrentUser()
            setUser(currentUser)
        } catch {
            logout()
        }
    }

    function logout() {
        clearAuth()
    }

    return {
        user,
        accessToken,
        refreshToken,
        isAuthenticated,
        login,
        logout,
        fetchCurrentUser,
        setTokens,
        clearAuth
    }
})

const validateForm = (): boolean => {
  errors.name = ''
  errors.email = ''
  errors.phone = ''
  errors.birthDate = ''

  let isValid = true

  if (!form.name.trim()) {
    errors.name = 'Введіть ім’я.'
    isValid = false
  }

  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

  if (!form.email.trim()) {
    errors.email = 'Введіть email.'
    isValid = false
  } else if (!emailRegex.test(form.email)) {
    errors.email = 'Введіть коректний email.'
    isValid = false
  } else {
    const emailExists = props.participants.some(
        participant =>
            participant.email.toLowerCase() === form.email.trim().toLowerCase()
    )

    if (emailExists) {
      errors.email = 'Учасник з таким email вже зареєстрований.'
      isValid = false
    }
  }

  const phoneRegex = /^\+380\d{9}$/

  if (!form.phone.trim()) {
    errors.phone = 'Введіть номер телефону.'
    isValid = false
  } else if (!phoneRegex.test(form.phone)) {
    errors.phone = 'Телефон має бути у форматі +380XXXXXXXXX.'
    isValid = false
  }

  if (!form.birthDate) {
    errors.birthDate = 'Виберіть дату народження.'
    isValid = false
  } else {
    const selectedDate = new Date(`${form.birthDate}T00:00:00`)
    const today = new Date()

    today.setHours(0, 0, 0, 0)

    if (selectedDate > today) {
      errors.birthDate = 'Дата народження не може бути в майбутньому.'
      isValid = false
    }
  }

  return isValid
}

const resetForm = () => {
  form.name = ''
  form.email = ''
  form.phone = ''
  form.birthDate = ''

  errors.name = ''
  errors.email = ''
  errors.phone = ''
  errors.birthDate = ''
}

const submitForm = () => {
  if (!validateForm()) {
    return
  }

  emit('add-participant', {
    name: form.name.trim(),
    email: form.email.trim(),
    phone: form.phone.trim(),
    birthDate: form.birthDate
  })

  resetForm()
}

const BASE_URL = import.meta.env.VITE_API_URL || 'https://dummyjson.com'

type TokensRefreshedCallback = (tokens: { accessToken: string; refreshToken: string }) => void

let onTokensRefreshedCallback: TokensRefreshedCallback | null = null
let refreshPromise: Promise<string> | null = null

export function onTokensRefreshed(cb: TokensRefreshedCallback) {
    onTokensRefreshedCallback = cb
}

function clearSessionAndRedirect() {
    localStorage.removeItem('access_token')
    localStorage.removeItem('refresh_token')
    localStorage.removeItem('auth_user')
    const baseUrl = import.meta.env.BASE_URL || '/'
    const loginUrl = baseUrl.endsWith('/') ? `${baseUrl}login` : `${baseUrl}/login`
    window.location.href = loginUrl
}

async function performTokenRefresh(): Promise<string> {
    const refreshToken = localStorage.getItem('refresh_token')

    if (!refreshToken) {
        clearSessionAndRedirect()
        throw new Error('No refresh token available')
    }

    try {
        const refreshResponse = await fetch(`${BASE_URL}/auth/refresh`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ refreshToken, expiresInMins: 60 })
        })

        if (!refreshResponse.ok) {
            clearSessionAndRedirect()
            throw new Error('Failed to refresh token')
        }

        const data = await refreshResponse.json()
        localStorage.setItem('access_token', data.accessToken)
        localStorage.setItem('refresh_token', data.refreshToken)

        if (onTokensRefreshedCallback) {
            onTokensRefreshedCallback({
                accessToken: data.accessToken,
                refreshToken: data.refreshToken
            })
        }

        return data.accessToken
    } catch (err) {
        clearSessionAndRedirect()
        throw err
    }
}

export async function http<T>(
    path: string,
    options: RequestInit = {}
): Promise<T> {
    let token = localStorage.getItem('access_token')

    const headers: Record<string, string> = {
        'Content-Type': 'application/json',
        ...(options.headers as Record<string, string>)
    }

    if (token) {
        headers['Authorization'] = `Bearer ${token}`
    }

    let response = await fetch(`${BASE_URL}${path}`, {
        ...options,
        headers
    })

    if (response.status === 401 && !path.includes('/auth/login') && !path.includes('/auth/refresh')) {
        if (!refreshPromise) {
            refreshPromise = performTokenRefresh().finally(() => {
                refreshPromise = null
            })
        }

        try {
            token = await refreshPromise
            headers['Authorization'] = `Bearer ${token}`
            response = await fetch(`${BASE_URL}${path}`, {
                ...options,
                headers
            })
        } catch {
            throw new Error('Unauthorized')
        }
    }

    if (!response.ok) {
        const errorData = await response.json().catch(() => ({}))
        throw new Error(errorData.message || `HTTP ${response.status}: ${response.statusText}`)
    }

    return response.json() as Promise<T>
}

    if (sortField.value === 'birthDate') {
      comparison =
          first.birthDate.localeCompare(second.birthDate)
    }

    return sortDirection.value === 'asc'
        ? comparison
        : -comparison
  })
})

  if (sortField.value === field) {
    sortDirection.value =
        sortDirection.value === 'asc'
            ? 'desc'
            : 'asc'

    return
  }

  sortField.value = field
  sortDirection.value = 'asc'
}

const getSortIcon = (field: SortField) => {
  if (sortField.value !== field) {
    return ''
  }

  return sortDirection.value === 'asc'
      ? ' ↑'

}

  }

  if (!editForm.value.name.trim()) {

    isValid = false
  }

  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

  if (!editForm.value.email.trim()) {
    editErrors.value.email = 'Введіть email.'
    isValid = false
  } else if (!emailRegex.test(editForm.value.email)) {
    editErrors.value.email = 'Введіть коректний email.'
    isValid = false
  } else {
    const emailExists = props.participants.some(
        participant =>
            participant.id !== participantToEdit.value?.id &&
            participant.email.toLowerCase() ===
            editForm.value.email.trim().toLowerCase()
    )

    if (emailExists) {
      editErrors.value.email =
          'Учасник з таким email вже зареєстрований.'
      isValid = false
    }
  }

  const phoneRegex = /^\+380\d{9}$/

  if (!editForm.value.phone.trim()) {
    editErrors.value.phone =
        'Введіть номер телефону.'
    isValid = false
  } else if (!phoneRegex.test(editForm.value.phone)) {

import { computed, ref } from 'vue'
import Winner from './Winner.vue'
import BaseButton from '../../../shared/ui/AppButton/BaseButton.vue'
import type { Participant } from '@/shared/config'

const props = defineProps<{
  participants: Participant[]
}>()

const emit = defineEmits<{
  'remove-winner': [id: string]
}>()

const winners = ref<Participant[]>([])

const canChooseWinner = computed(() => {
  return (
      props.participants.length > 0 &&
      winners.value.length < 3 &&
      winners.value.length < props.participants.length
  )
})

const chooseNewWinner = () => {
  if (!canChooseWinner.value) {
    return
  }

  const availableParticipants = props.participants.filter(
      participant =>
          !winners.value.some(
              winner => winner.id === participant.id
          )
  )

  if (availableParticipants.length === 0) {
    return
  }

  const randomIndex = Math.floor(
      Math.random() * availableParticipants.length
  )

  const winner = availableParticipants[randomIndex]

  if (winner) {
    winners.value.push(winner)
  }
}

const removeWinner = (id: string) => {
  winners.value = winners.value.filter(
      winner => winner.id !== id
  )

  emit('remove-winner', id)
}

const removeParticipantFromWinners = (id: string) => {
  winners.value = winners.value.filter(
      winner => winner.id !== id
  )
}

defineExpose({
  removeParticipantFromWinners
})
</script>

      >
        <Winner
            :winner="winner"
